# UX polish Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** System screens (404, error, 403, waiting), Polish UI and metadata, loading skeletons, empty states and a role-aware home page.

**Architecture:** One `StatusScreen` component (+ SVG illustration driven by theme CSS variables) serves all system states. Skeletons come from a shadcn-style `Skeleton`, wired through route `loading.tsx` files. The home page composes existing domain queries (`listPackagesForViewer`, `listReports`) through a pure, unit-tested `summarizeDashboard`.

**Tech Stack:** Next.js 15.5 (App Router), React 19, Tailwind 4, shadcn/ui, lucide-react, Clerk, Vitest.

## Global Constraints

From [the spec](../specs/2026-10-20-ux-polish-design.md):

- System screens are ALWAYS one vertical column at every width, in this order: illustration, big code, text 1 (title), text 2 (description), button(s).
- Illustration is inline SVG styled only with theme CSS variables (`--background`, `--muted-foreground`, `--primary`, `--destructive`, `--border`), so all palettes and light/dark work. The lit window pulses with `motion-safe:animate-pulse` only.
- Polish only, no i18n library; `<html lang="pl">`; code identifiers, DB values and enum names stay English.
- Error screens never show `error.message`; only a muted `error.digest`.
- No new npm dependencies. No global top progress bar.
- Dates keep `dd.MM.yyyy HH:mm`.
- Do not change report/package domain logic. Do not touch `lib/generated`.

## Working notes

- Execute in the worktree `.claude/worktrees/ux-polish` (branch `feat/ux-polish`); `node_modules` is symlinked and `.env` copied. Commit with explicit `git add` paths, never `-A`.

export type StatusVariant = "not-found" | "error" | "forbidden" | "waiting";

const fill = (variable: string, opacity?: number) => ({
  fill: `var(${variable})`,
  fillOpacity: opacity,
});

const LIT = "#f6c445";

export function StatusIllustration({ variant }: { variant: StatusVariant }) {
  const lit = variant === "not-found" || variant === "waiting";
  const windows: [number, number][] = [
    [64, 34], [100, 34], [136, 34],
    [64, 62], [136, 62],
    [64, 90], [136, 90],
  ];
  return (
    <svg viewBox="0 0 220 150" className="h-auto w-44" aria-hidden="true">
      <rect x="30" y="134" width="160" height="4" rx="2" style={fill("--border")} />
      <rect x="52" y="22" width="116" height="112" rx="6" style={fill("--muted-foreground", 0.3)} />
      {windows.map(([x, y]) => (
        <rect key={`${x}-${y}`} x={x} y={y} width="20" height="18" rx="2" style={fill("--background")} />
      ))}
      <rect
        x="100"
        y="62"
        width="20"
        height="18"
        rx="2"
        className={lit ? "motion-safe:animate-pulse" : undefined}
        style={lit ? { fill: LIT } : fill("--muted-foreground", 0.45)}
      />
      {variant === "error" && (
        <path
          d="M104 66 l12 10 m0 -10 l-12 10"
          stroke="var(--destructive)"
          strokeWidth="2.5"
          strokeLinecap="round"
          fill="none"
        />
      )}
      <rect x="96" y="96" width="28" height="38" rx="3" style={fill("--muted-foreground", 0.75)} />
      {variant === "waiting" && (
        <path d="M96 96 L108 100 L108 134 L96 134 Z" style={fill("--foreground", 0.35)} />
      )}
      {variant === "forbidden" && (
        <g>
          <rect x="104" y="108" width="12" height="10" rx="2" style={{ fill: LIT }} />
          <path
            d="M107 108 v-3 a3 3 0 0 1 6 0 v3"
            fill="none"
            stroke={LIT}
            strokeWidth="2"
          />
        </g>
      )}
    </svg>
  );
}
```

import type { PackageStatus, ReportStatus } from "@/lib/generated/prisma";

const LIST_LIMIT = 3;
const OPEN_STATUSES: ReportStatus[] = ["NEW", "IN_PROGRESS"];

export function summarizeDashboard<
  P extends { status: PackageStatus; receivedAt: Date },
  R extends { status: ReportStatus; urgent: boolean; authorId: string; createdAt: Date },
>(input: { userId: string; packages: P[]; reports: R[] }) {
  const waiting = input.packages
    .filter((p) => p.status !== "PICKED_UP")
    .sort((a, b) => b.receivedAt.getTime() - a.receivedAt.getTime());

  const mine = input.reports
    .filter((r) => r.authorId === input.userId && OPEN_STATUSES.includes(r.status))
    .sort((a, b) => b.createdAt.getTime() - a.createdAt.getTime());

  return {
    waitingPackages: waiting.slice(0, LIST_LIMIT),
    waitingPackageCount: waiting.length,
    myOpenReports: mine.slice(0, LIST_LIMIT),
    myOpenReportCount: mine.length,
    newReportCount: input.reports.filter((r) => r.status === "NEW").length,
    urgentReportCount: input.reports.filter((r) => r.urgent && OPEN_STATUSES.includes(r.status)).length,
  };
}
```
